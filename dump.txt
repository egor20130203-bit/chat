import { useEffect, useRef, useState } from 'react';
import { QUALITY } from '../hooks/useCall';

const Icon = ({ name, size = 20 }) => (
  <svg viewBox="0 0 24 24" width={size} height={size} fill="currentColor" aria-hidden="true">
    <path d={PATHS[name]} />
  </svg>
);

export default function CallOverlay({ callApi, peerName, myName, deafened, volume = 100 }) {
  const {
    call,
    remoteStream,
    remoteVideoOn,
    localVideo,
    micOn,
    cameraOn,
    screenOn,
    quality,
    stats,
    acceptCall,
    endCall,
    toggleMic,
    toggleCamera,
    toggleScreen,
    setQuality,
  } = callApi;

  const remoteRef = useRef(null);
  const localRef = useRef(null);
  const audioRef = useRef(null);
  const boxRef = useRef(null);
  const dragRef = useRef(null);

  const [soundBlocked, setSoundBlocked] = useState(false);
  const [minimized, setMinimized] = useState(false);
  const [pos, setPos] = useState(null); // null = стоит в углу по умолчанию

  useEffect(() => {
    if (remoteRef.current && remoteRef.current.srcObject !== remoteStream) {
      remoteRef.current.srcObject = remoteStream ?? null;
    }
  }, [remoteStream, remoteVideoOn]);

  useEffect(() => {
    if (localRef.current) localRef.current.srcObject = localVideo ?? null;
  }, [localVideo]);

  // Звук собеседника идёт отдельным элементом — так надёжнее
  useEffect(() => {
    const el = audioRef.current;
    if (!el) return;
    if (el.srcObject !== remoteStream) el.srcObject = remoteStream ?? null;
    if (!remoteStream) return;

    const id = localStorage.getItem('chat.spk');
    if (id && typeof el.setSinkId === 'function') {
      el.setSinkId(id).catch((e) => console.warn('Наушники не выбрались, играем в устройство по умолчанию:', e));
    }

    el.volume = Math.min(1, Math.max(0, volume / 100));
    el.muted = !!deafened;
    el.play()
      .then(() => setSoundBlocked(false))
      .catch((e) => {
        console.warn('Браузер не дал включить звук:', e);
        setSoundBlocked(true);
      });
  }, [remoteStream, deafened, volume]);

  useEffect(() => {
    if (!audioRef.current) return;
    audioRef.current.muted = !!deafened;
    audioRef.current.volume = Math.min(1, Math.max(0, volume / 100));
  }, [deafened, volume]);

  // Входящий звонок всегда разворачиваем, чтобы не пропустить
  useEffect(() => {
    if (call?.status === 'incoming') setMinimized(false);
  }, [call?.status]);

  // Новый звонок — окно снова в углу
  useEffect(() => {
    if (!call) {
      setPos(null);
      setMinimized(false);
    }
  }, [call]);

  // Если окно вынесли за край (уменьшили браузер) — возвращаем в видимую зону
  useEffect(() => {
    const onResize = () => {
      setPos((p) => {
        if (!p) return p;
        const el = boxRef.current;
        if (!el) return p;
        return {
          x: Math.min(p.x, Math.max(0, window.innerWidth - el.offsetWidth)),
          y: Math.min(p.y, Math.max(0, window.innerHeight - el.offsetHeight)),
        };
      });
    };
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, []);

  const unmute = async () => {
    try {
      await audioRef.current?.play();
      setSoundBlocked(false);
    } catch (e) {
      alert('Не удалось включить звук: ' + e.message);
    }
  };

  const onHeadPointerDown = (e) => {
    if (e.target.closest('button')) return; // по кнопкам не таскаем
    const el = boxRef.current;
    if (!el) return;
    const r = el.getBoundingClientRect();
    dragRef.current = { dx: e.clientX - r.left, dy: e.clientY - r.top };
    setPos({ x: r.left, y: r.top });
    try {
      e.currentTarget.setPointerCapture(e.pointerId);
    } catch {}
  };

  const onHeadPointerMove = (e) => {
    const d = dragRef.current;
    if (!d) return;
    const el = boxRef.current;
    const w = el?.offsetWidth ?? 0;
    const h = el?.offsetHeight ?? 0;
    setPos({
      x: Math.min(Math.max(0, e.clientX - d.dx), Math.max(0, window.innerWidth - w)),
      y: Math.min(Math.max(0, e.clientY - d.dy), Math.max(0, window.innerHeight - h)),
    });
  };

  const onHeadPointerUp = (e) => {
    dragRef.current = null;
    try {
      e.currentTarget.releasePointerCapture(e.pointerId);
    } catch {}
  };

  if (!call) return null;

  const incoming = call.status === 'incoming';
  const active = call.status === 'active';
  const status = incoming ? 'Входящий звонок' : active ? '' : 'Вызываем…';

  const boxStyle = pos ? { position: 'fixed', left: pos.x, top: pos.y, margin: 0 } : undefined;

  return (
    <div className="dc-backdrop">
      <div
        ref={boxRef}
        className={minimized ? 'dc-call mini' : 'dc-call'}
        style={boxStyle}
      >
        <div
          className="dc-head"
          onPointerDown={onHeadPointerDown}
          onPointerMove={onHeadPointerMove}
          onPointerUp={onHeadPointerUp}
          onPointerCancel={onHeadPointerUp}
        >
          <span className="dc-head-name">{peerName}</span>
          {status && <span className="dc-head-status">{status}</span>}
          {!incoming && (
            <button
              className="dc-head-btn"
              onClick={() => setMinimized((v) => !v)}
              title={minimized ? 'Развернуть' : 'Свернуть'}
            >
              <Icon name={minimized ? 'expand' : 'minimize'} size={16} />
            </button>
          )}
        </div>

        <audio ref={audioRef} autoPlay playsInline />
        {soundBlocked && (
          <button className="dc-unmute" onClick={unmute}>
            Нажмите, чтобы включить звук
          </button>
        )}

        <div className="dc-stage">
          {/* Собеседник */}
          <div className="dc-tile">
            {remoteVideoOn ? (
              <video ref={remoteRef} autoPlay playsInline muted className="dc-video" />
            ) : (
              <>
                <div className={incoming || !active ? 'dc-avatar ringing' : 'dc-avatar'}>
                  {initials(peerName)}
                </div>
                <div className="dc-tile-name">{peerName}</div>
              </>
            )}
          </div>

          {/* Ты */}
          <div className="dc-tile">
            {localVideo ? (
              <video
                ref={localRef}
                autoPlay
                playsInline
                muted
                className={screenOn ? 'dc-video' : 'dc-video mirrored'}
              />
            ) : (
              <>
                <div className="dc-avatar self">{initials(myName)}</div>
                <div className="dc-tile-name">{myName}</div>
              </>
            )}
            {localVideo && stats && (
              <div className="dc-stat">
                {stats.w}×{stats.h} · {stats.fps} fps
              </div>
            )}
          </div>
        </div>

        <div className="dc-pill">
          {incoming ? (
            <>
              <button className="dc-btn accept" onClick={acceptCall} title="Ответить">
                <Icon name="call" />
              </button>
              <button className="dc-btn hangup" onClick={endCall} title="Отклонить">
                <Icon name="end" />
              </button>
            </>
          ) : (
            <>
              <button className={micOn ? 'dc-btn' : 'dc-btn off'} onClick={toggleMic} title="Микрофон">
                <Icon name={micOn ? 'mic' : 'micOff'} />
              </button>
              <button
                className={cameraOn ? 'dc-btn on' : 'dc-btn'}
                onClick={toggleCamera}
                disabled={!active}
                title="Камера"
              >
                <Icon name={cameraOn ? 'cam' : 'camOff'} />
              </button>
              <button
                className={screenOn ? 'dc-btn on' : 'dc-btn'}
                onClick={toggleScreen}
                disabled={!active}
                title="Демонстрация экрана"
              >
                <Icon name="screen" />
              </button>
              <select
                className="dc-quality"
                value={quality}
                onChange={(e) => setQuality(e.target.value)}
                title="Качество"
              >
                {Object.entries(QUALITY).map(([k, v]) => (
                  <option key={k} value={k}>
                    {v.label}
                  </option>
                ))}
              </select>
              <button className="dc-btn hangup" onClick={endCall} title="Завершить">
                <Icon name="end" />
              </button>
            </>
          )}
        </div>
      </div>
    </div>
  );
}
